\begin{notepanels}[columns=2,widths={2,1},gap=6mm]
  \begin{notepanel}{Model}
    A first-order low-pass filter has one time constant.
    Its output follows slow changes and smooths rapid changes.
    \[H(s)=\frac{1}{1+sRC}\]
  \end{notepanel}
  \begin{notepanel}{Interpretation}
    Larger $RC$ means a slower response and a lower cutoff.
  \end{notepanel}
\end{notepanels}
