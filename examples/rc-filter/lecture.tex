% ==================== Topic 1: frequency domain ====================
\notesection{RC low-pass filter: frequency domain}
A resistor and capacitor smooth a changing input. Take the output across
the capacitor: slow changes pass through, while fast changes are attenuated.

\assumption{Ideal voltage source, ideal components, and no output load.
For the step response, the capacitor is initially uncharged.}

\keyequation{Transfer function and cutoff}{
  H(s)=\frac{V_{\mathrm{out}}(s)}{V_{\mathrm{in}}(s)}
      =\frac{1}{1+sRC},\qquad
  \tau=RC,\qquad f_c=\frac{1}{2\pi RC}
}

\begin{notetable}{lXr}{\textbf{Quantity} & \textbf{Meaning} & \textbf{Value}}
$R$ & Series resistance & $1.0\,\mathrm{k}\Omega$ \\
$C$ & Shunt capacitance & $100\,\mathrm{nF}$ \\
$\tau$ & Time constant & $100\,\mu\mathrm{s}$ \\
$f_c$ & Half-power cutoff frequency & $1.59\,\mathrm{kHz}$ \\
\end{notetable}

\begin{notecircuit}
  \draw (0,0) node[ground]{} to[V,l=$v_{\mathrm{in}}$] (0,2)
    to[R,l=$R$] (3,2) to[short,-o] (4.2,2)
    node[right]{$v_{\mathrm{out}}$};
  \draw (3,2) to[C,l=$C$,*-] (3,0) node[ground]{};
\end{notecircuit}

% ====================== Topic 2: time domain =======================
\notesection{Step response: time domain}
\keyequation{Response to a step of amplitude $V_0$}{
  v_{\mathrm{out}}(t)=V_0\!\left(1-e^{-t/\tau}\right),\qquad t\geq 0
}

\begin{notesignal}[
  xlabel={Time $t/\tau$},ylabel={$v_{\mathrm{out}}/V_0$},
  xmin=0,xmax=5,ymin=0,ymax=1.08,
  xtick={0,1,2,3,4,5},ytick={0,0.5,1},
  height=4.6cm]
  \addplot[domain=0:5,samples=100] {1-exp(-x)};
  \addplot[only marks,mark=*,mark size=2pt] coordinates {(1,0.63212)};
  \node[anchor=north west,font=\small,text=NoteAccent]
    at (axis cs:0.25,1.02) {$63.2\%$ at $t=\tau$};
\end{notesignal}

\textbf{Frequency conventions.} At DC, the gain tends to one. At the cutoff,
$|H|=1/\sqrt{2}$ and the phase is $-45^\circ$.
Use $\omega_c=1/RC$ for rad/s and $f_c=1/(2\pi RC)$ for Hz.

% =================== Topic 3: sampled signals ======================
\notesection{Sampled signals: definitions and notation}

\notedefinition{Sampling period}{The interval $T_s$ between successive
samples. The sampling frequency is $f_s=1/T_s$. A sampled input is written
$x[n]=v_{\mathrm{in}}(nT_s)$, where $n$ is an integer sample index.}

\begin{notation}
$T_s$ & Sampling period & $\mathrm{s}$ \\
$f_s$ & Sampling frequency & $\mathrm{Hz}$ \\
$x[n]$ & Input voltage sample & $\mathrm{V}$ \\
$y[n]$ & Filtered output sample & $\mathrm{V}$ \\
$\alpha$ & Discrete filter coefficient & dimensionless \\
\end{notation}

% ================== Topic 4: model derivation ======================
\notesection{From the circuit equation to a recurrence}
Using a backward difference for the output derivative gives a simple
discrete approximation of the continuous RC model, with $\tau=RC$.

\begin{notederivation}{Backward-Euler discretization}
  \tau\frac{dy}{dt}+y &= x
    && \text{continuous model} \\
  \tau\frac{y[n]-y[n-1]}{T_s}+y[n] &= x[n]
    && \text{backward difference} \\
  (\tau+T_s)y[n] &= T_s x[n]+\tau y[n-1]
    && \text{collect terms} \\
  y[n] &= y[n-1]+\alpha\bigl(x[n]-y[n-1]\bigr)
    && \alpha=\frac{T_s}{\tau+T_s}
\end{notederivation}

% =================== Topic 5: embedded code ========================
\notesection{Code fragments from the lecture}
\textbf{C: one sample update.} Store the return value as the next state.

\begin{notecode}[language=C]
float lowpass_step(float x, float previous_y, float alpha)
{
    return previous_y + alpha * (x - previous_y);
}
\end{notecode}

\textbf{ARM Thumb:} set bit 5 in \texttt{r1}; \texttt{r0} is a temporary register.

\begin{notecode}[language=ARM]
MOVS r0, #1       ; Start with a single set bit
LSLS r0, r0, #5   ; Move it to bit 5
ORRS r1, r1, r0   ; Set that bit in the flags word
\end{notecode}
